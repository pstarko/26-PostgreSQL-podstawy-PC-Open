--Laczenie tabel

--kartezjan
select *
	from employees, departments
order by 1

--połączenie wewnętrze z kartezjana
select 
	e.last_name,
	d.department_name
	from employees e, departments d
where 
	e.department_id=d.department_id


--łączenie wewnętrze
select 
	e.last_name,
	d.department_name
	from employees e JOIN departments d on e.department_id=d.department_id
	
--3 tabele
--połączenie wewnętrze z kartezjana
select 
	e.last_name,
	d.department_name,
	l.city 
	from employees e, departments d, locations l 
where 
	e.department_id=d.department_id
	and d.location_id = l.location_id 


--łączenie wewnętrze
select 
	e.last_name,
	d.department_name,
	l.city 
	from employees e JOIN departments d on e.department_id=d.department_id
					 join locations l on d.location_id = l.location_id 
					 
					 
--łączenie lewostronne zewnętrze
select 
	e.last_name,
	d.department_name
	from employees e left JOIN departments d on e.department_id=d.department_id					 
		
	
--łączenie prawostronne zewnętrze
select 
	e.last_name,
	d.department_name
	from employees e right JOIN departments d on e.department_id=d.department_id		

--łączenie obustronne zewnętrze (JOIN + LEFT + RIGHT)
select 
	e.last_name,
	d.department_name
	from employees e full JOIN departments d on e.department_id=d.department_id		
	
	
--selfjoin
select
	e.employee_id, 
	e.last_name,
	b.employee_id,
	b.last_name
from employees e join employees b on e.employee_id=b.manager_id

--antyjoin
select 
	e.last_name,
	e.department_id,
	d.department_id
from employees e join departments d 	on e.department_id != d.department_id
order by 1,3


--połączenia wierszowe
--bez powtórzeń
select department_id from employees e 
union
Select department_id from departments d 

--z duplikatami
select department_id from employees e 
union all
Select department_id from departments d 

--cześć wspólna
select department_id from employees e 
intersect
select department_id from departments d
 
--różnica
select department_id from departments d
except
select department_id from employees e
