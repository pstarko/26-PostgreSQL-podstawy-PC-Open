/*
1. Wyświetl najniższą i najwyższą pensję w firmie. Wyświetl informacje o różnicy między
najlepiej i najgorzej opłacanymi pracownikami (1 wiersz).
*/

select
	max(salary) maxi,
	min(salary) mini,
	max(salary) - min(salary) diff
	from employees

/*
2. Wyświetl średnie wynagrodzenie dla wszystkich job_id. Wyklucz zespoły 10,20 i 50.
Sortuj wyniki według malejącej średniej płacy (13 wierszy).

*/

select
	job_id,
	avg(salary) srednia
	from employees
where
	department_id in (10,20,50)
group by 
	job_id
order by 	
	srednia desc


/*
3. Wyświetl liczbę IT_PROG zatrudnionych w firmie (1 wiersz)


*/

select
	count(*)
	from employees
where
	job_id = 'IT_PROG'


/*
4. Znajdź łączne miesięczne wynagrodzenie dla każdego działu. Nie zapomnij o
dodatkowych zarobkach. W wyniku usuń wartość null i dane posortuj w taki sposób
by zespół 110 był pierwszy, a reszta już zgodnie z wartością indeksu(11 wierszy)

*/

select
	department_id,
	sum(salary + (salary * coalesce(commission_pct,0))) suma
	from employees
where
	department_id is not null
group by 
	department_id
order by
	case
		when department_id = 110 then 9
		else department_id
	end

/*
5. Dla każdego szefa wyświetl pensję najgorzej zarabiającego podwładnego. Pomiń
pracowników zarabiających mniej niż 5000 i nie mających szefa (12 wierszy).
*/

select
	manager_id,
	min(salary)
	from employees e
where
	salary > 5000
	and manager_id is not null
group by 
	manager_id
/*
6. Sprawdź, ilu mamy pracowników z literą O i R w nazwisku. Napisz zapytanie odporne
na wielkość liter w tabeli (1 wiersz)

*/

select 
	count(*)
	from  employees e 
where
	lower(last_name) like '%o%'
	and lower(last_name) like '%r%'

	
select
		count(*) licznik
	from employees e 
	where
		position('o' in lower(last_name)) > 0
		and position('r' in lower(last_name)) > 0	
		
/*
7. Czy istnieje dział, który będzie miesięcznie zarabiał ponad 52673 (salary), gdy otrzyma
10 razy 10% podwyżki? Pokaż identyfikator tego działu i kwotę zaokrągloną do liczb
całkowitych w górę. (7 wierszy)
*/
		
select
	department_id,
	ceiling(sum(Salary) * power(1.1, 10)) result
	from employees
group by 
	department_id
having
	sum(Salary) * power(1.1, 10) > 52673


/*
9. Wyświetl numery zespołów wraz z liczbą płac dodatkowych w każdym dziale. Sortuj
wynik po malejącym id_działu (11 wierszy).

*/

select 
	department_id ,
	count(commission_pct)
	from  employees 
where
	department_id is not null
group by 	
	department_id
order by
	department_id desc


		
		
/*
10. Zmodyfikuj zapytanie w zadaniu 9, aby wyświetlić liczby tylko tych działów, które
zatrudniają więcej niż 10 pracowników (2 wiersze).
