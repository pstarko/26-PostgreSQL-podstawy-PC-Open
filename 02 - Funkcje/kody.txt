--Funkcje

--warunkowe
SELECT
	last_name,
	salary,
	CASE
		WHEN salary > 10000 THEN 'OK'
	ELSE '---'
	END warunek
FROM
	employees

SELECT
	last_name,
	salary,
	department_id,
	CASE
		WHEN salary > 10000 and department_id = 90 THEN 'OK'
	ELSE '---'
	END warunek
FROM
	employees
	
SELECT
	last_name,
	salary,
	CASE
		WHEN salary > 17000 THEN '>17k'
		WHEN salary > 10000 THEN '>10k'
	ELSE '---'
	END warunek
FROM
	employees
	

--ciągi znaków
	
SELECT
	last_name,
	lower(last_name) lower,
	upper(last_name) upper,
	job_id,
	InitCap(job_id) ic 
FROM
	employees
	
	
SELECT
	last_name,
	substring(last_name, 1, 2) substr,
	substring(last_name, 2, 3) substr2
FROM
   employees
   
	
-- w postgres substring nie przyjmuje negatywnych argumentow
SELECT
	last_name,
	substring(last_name, -2, 2) substr
FROM
   employees	
	
	
SELECT
	last_name,
	replace(last_name, 'in','XYZ') replace1,
	replace(last_name, 'a','WWW') replace2,
	translate(last_name, 'in', 'YX') translate1,
	translate(last_name, 'in', '') translate2
FROM
	employees

SELECT
	last_name,
	REVERSE(last_name) reverse,
	RIGHT(last_name, 2) r,
	LEFT(last_name, 2) l
FROM
	employees

SELECT 
	'spacja ' ze_spacja,
	trim('spacja ') t
	
SELECT 
	last_name ,
	rtrim(last_name, 'l') rt,
	ltrim(last_name, 'A') lt,
	lpad(last_name, 10, 'X'),
	rpad(last_name, 10, 'X')
FROM employees e
order by last_name

select
	last_name,
	position('a' in last_name),
	position('g' in last_name) 
FROM employees 


SELECT
	last_name,
	length(last_name) length 
FROM
	employees

select 
	first_name ,
	last_name ,
	concat(first_name, ' ', last_name) konkatenacja1,
	first_name || ' ' || last_name konkatenacja2
from employees e

	DATE_PART('year',hire_date) rok,

SELECT
	hire_date,
	age(hire_date)
from employees


SELECT
	hire_date,
	hire_date + interval '1 year' dodany_rok,
	hire_date + interval '1 month' dodany_miesiac,
	(hire_date + interval '1 day')::Date dodany_dzien
from employees

select
	hire_date,
	to_char(hire_date, 'DD-Month-YYYY') format1,
	to_char(hire_date, 'Day, DD-MM-YY') format2
from employees e
