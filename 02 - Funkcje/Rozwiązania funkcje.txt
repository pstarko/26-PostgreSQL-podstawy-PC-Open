/*
1. Dla każdego pracownika wygeneruj kod składający się z pierwszych dwóch liter job_id
oraz całego employee_id

*/
select
	job_id,
	employee_id,
	left(job_id, 2) || employee_id code,
	CONCAT(substring(job_id, 1, 2), employee_id) code2
	from employees

/*
2. Wydaj wojnę literom l, m oraz k i zamień je na X. Użyj funkcji REPLACE i TRANSLATE.
*/

select 
	last_name,
	translate(last_name , 'lkm', 'XXX'),
	replace(
		replace(	
			REPLACE(last_name , 'l', 'X')
									, 'k', 'X')
										, 'm','X') repla
	from employees
