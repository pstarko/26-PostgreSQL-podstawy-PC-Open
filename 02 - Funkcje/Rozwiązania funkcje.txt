/*
1. Dla każdego pracownika wygeneruj kod składający się z pierwszych dwóch liter job_id
oraz całego employee_id

*/
select
	job_id,
	employee_id,
	left(job_id, 2) || employee_id code,
	CONCAT(substring(job_id, 1, 2), employee_id) code2
	from employees

/*
2. Wydaj wojnę literom l, m oraz k i zamień je na X. Użyj funkcji REPLACE i TRANSLATE.
*/

select 
	last_name,
	translate(last_name , 'lkm', 'XXX'),
	replace(
		replace(	
			REPLACE(last_name , 'l', 'X')
									, 'k', 'X')
										, 'm','X') repla
	from employees

/*
3. Dodaj podwyżkę 15% dla każdego pracownika. Uwzględnij również
dodatek(commission_pct). Posortuj malejąco po podniesionej płacy

*/
select
	last_name,
	(salary + (salary * coalesce(commission_pct ,0))) * 1.15 more_money
	from employees
order by
	more_money desc

/*
4. Policz ile dni jest zatrudniony każdy pracownik. Wyróżnij odpowiednim
komentarzem tych, którzy pracują ponad 3500 dni (Wynik na dzień 24-04-2025)
*/

select 
	last_name,
	'2025-04-24'::date - hire_date diff,
	case 
		when '2025-04-24'::date - hire_date > 3500 then 'Pracuje więcej niż 3500 dni'
		else '---------'
	end	result
	from employees


/*
Rozwiąż wyrażenie algebraiczne: Resztę z dzielenia 134 przez 8 przemnóż przez
pierwiastek z 81. Podziel to wszystko przez 244 i podnieś do 4 potęgi. Do wyniku
dodaj 0.0097. Zaogrąglij wszystko do 4 miejsc po przecinku zgodnie z zasadami
matematyki.

*/

select
	round((power((134%8 * sqrt(81)) / 244, 4) + 0.0097)::numeric, 4)


/*
6. Pokaż last name w taki sposób, by ostatnie trzy litery były w odwróconej kolejności

*/
--1
select 
	last_name,
	left(last_name, -3),
	reverse(right(last_name, 3)),
	left(last_name, -3) || reverse(right(last_name, 3)) code
	from employees

	
--2	
select 
	last_name ,
	trim(last_name, right(last_name , 3 )),
	reverse(right(last_name , 3)),
	trim(last_name, right(last_name , 3 ))  || reverse(right(last_name , 3)) lastname
from employees e


--3
select 
	last_name, 
	left(last_name,-3),
	reverse(right(last_name,3)),
	concat(left(last_name,-3), reverse(right(last_name,3))) as code 
from employees e


--4
select
	last_name,
	substring(last_name, 1, case 
								when length(last_name) - 3 < 0 then 0 
								else length(last_name) - 3 end),
	substring(last_name, 1, case 
								when length(last_name) - 3 < 0 then 0 
								else length(last_name) - 3 end) ||
	reverse(right(last_name, 3))
	from employees
