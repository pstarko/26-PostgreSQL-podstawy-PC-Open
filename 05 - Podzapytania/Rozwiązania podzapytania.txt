/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Miller (5 wierszy)

*/

select 
	e.first_name ,
	e.last_name ,
	(select job_title from jobs where job_id = e.job_id ) jt
	from employees e
where
	e.department_id = (select department_id from employees e2 
						where e2.last_name ='Miller')	


/*
2. Wyświetl następujące dane o najdłużej zatrudnionym pracowniku na stanowisku
FI_ACCOUNT (1 wiersz)

*/

select 
	e.employee_id ,
	e.first_name ,
	e.last_name ,
	e.hire_date ,
	e.job_id 
	from employees e
where
	e.hire_date = (select min(hire_date) from employees
					where job_id = 'FI_ACCOUNT')
	and e.job_id = 'FI_ACCOUNT'

--109 -> 2012-08-16 -> FI_ACCOUNT
--100 -> 2013-06-17 -> AD_PRES
					
update employees set hire_date = '2013-06-17' where employee_id  = 100


/*
3. Zobacz najdłużej pracujących pracowników każdego działu. (12 wierszy)

*/

select 
	e.first_name ,
	e.last_name ,
	e.department_id ,
	d.department_name ,
	e.hire_date 
from employees e join departments d on e.department_id = d.department_id
where
	(e.department_id , e.hire_date ) in (select department_id, min(hire_date) from employees
											group by department_id)


select 
	e.first_name ,
	e.last_name ,
	e.department_id ,
	d.department_name 
from employees e
	join departments d on e.department_id = d.department_id 
where e.hire_date = 
		(select
			min(e2.hire_date)
		from employees e2 
		where e2.department_id = e.department_id )


/*
4. Wyświetlaj te nazwy stanowisk, w których nie zatrudniono nikogo z działu
	MARKETING. Wykorzystaj podzapytanie w podzapytaniu (17 wierszy)

*/
explain
select
	*
	from jobs
where
	job_id in (select distinct job_id from employees
				where department_id not in (select department_id from departments
												where department_name ='Marketing'))
			

/*
6. Wyświetl nazwiska poniższe dane pracowników związanych ze sprzedażą i mających
innego szefa niż John Singh – wykorzystaj w zapytaniu strukturę „derrived column”
(29 wierszy)
*/


select
	e.last_name ,
	e.email ,
	e.department_id ,
	(select first_name || ' ' || last_name from employees e2 where e2.employee_id = e.manager_id ) boss_name	
	from employees e
where
	e.job_id  in (
					select
						job_id
						from jobs j 
					where lower(j.job_title ) like '%sales%')
    and e.manager_id  != (select employee_id from employees
    							where first_name = 'John' and last_name = 'Singh')
/*
7. Użyj skorelowanego podzapytania, aby wyświetlić informacje o pracowniku
niezatrudnionym w żadnym dziale (użyj właściwej formy operatora EXISTS) (1 wiersz)

8. Wyświetl nazwę działu, który płaci Twoim pracownikom najwięcej pieniędzy
miesięcznie(1 wiersz)

*/

--7
select
    e.first_name,
    e.last_name
    from employees e 
    where 
    		not exists (select 1 from departments dd where dd.department_id = e.department_id)
	
--8
select
	(select department_name from departments where department_id = e.department_id) dn,
	sum(salary)
	from employees e
group by 
	e.department_id 
having
	sum(salary) = (select max(t1.sum) from 
						(select sum(salary) sum from employees
							group by department_id) t1 )
